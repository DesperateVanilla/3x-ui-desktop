cmake_minimum_required(VERSION 3.24)
project(ThreeXControl VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network Sql Test)
add_library(fleet_core STATIC src/domain.h src/three_x_ui_api.h src/three_x_ui_api.cpp src/local_store.h src/local_store.cpp)
target_include_directories(fleet_core PUBLIC src)
target_link_libraries(fleet_core PUBLIC Qt6::Core Qt6::Network Qt6::Sql)
if(WIN32)
  target_link_libraries(fleet_core PUBLIC Crypt32)
endif()
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
  target_compile_options(fleet_core PRIVATE /utf-8 /W4 /permissive-)
endif()
qt_add_executable(ThreeXControl WIN32 src/main.cpp src/main_window.h src/main_window.cpp src/chart_widget.h src/chart_widget.cpp src/editor_dialogs.h src/editor_dialogs.cpp)
target_link_libraries(ThreeXControl PRIVATE fleet_core Qt6::Widgets)
set_target_properties(ThreeXControl PROPERTIES OUTPUT_NAME "3X-Control")
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(fleet_core_tests tests/core_tests.cpp)
  set_target_properties(fleet_core_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(fleet_core_tests PRIVATE fleet_core Qt6::Test)
  add_test(NAME fleet_core_tests COMMAND fleet_core_tests -o "${CMAKE_CURRENT_BINARY_DIR}/core-results.txt,txt" -o -,txt)
  set_tests_properties(fleet_core_tests PROPERTIES TIMEOUT 120)
  qt_add_executable(fleet_storage_tests tests/storage_tests.cpp)
  set_target_properties(fleet_storage_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(fleet_storage_tests PRIVATE fleet_core Qt6::Test)
  add_test(NAME fleet_storage_tests COMMAND fleet_storage_tests -o "${CMAKE_CURRENT_BINARY_DIR}/storage-results.txt,txt" -o -,txt)
  set_tests_properties(fleet_storage_tests PROPERTIES TIMEOUT 120)
  qt_add_executable(fleet_dialog_tests tests/dialog_tests.cpp src/editor_dialogs.h src/editor_dialogs.cpp)
  set_target_properties(fleet_dialog_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(fleet_dialog_tests PRIVATE fleet_core Qt6::Widgets Qt6::Test)
  add_test(NAME fleet_dialog_tests COMMAND fleet_dialog_tests -o "${CMAKE_CURRENT_BINARY_DIR}/dialog-results.txt,txt" -o -,txt)
  set_tests_properties(fleet_dialog_tests PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_FONTDIR=C:/Windows/Fonts")
endif()
